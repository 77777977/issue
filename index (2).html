직접 Node.js / Python 서버를 띄워 Signal.bz 및 네이버 공식 뉴스 API까지 연동하고 싶을 때 사용하는 서버 코드입니다. 
// [Node.js + Express] 자체 도메인 서버 구축용 실시간 검색어 & 뉴스 API
// 실행: npm install express axios rss-parser cors && node server.js
const express = require('express');
const axios = require('axios');
const Parser = require('rss-parser');
const cors = require('cors');

const app = express();
const parser = new Parser();
app.use(cors());
app.use(express.static('.')); // index.html 정적 호스팅

app.get('/api/realtime-keywords', async (req, res) => {
    try {
        const response = await axios.get('https://api.signal.bz/news/realtime');
        res.json({ top10: response.data.top10 });
    } catch (e) {
        res.status(500).json({ error: e.message });
    }
});

app.get('/api/news', async (req, res) => {
    const q = req.query.q || '속보';
    const feed = await parser.parseURL(`https://news.google.com/rss/search?q=${encodeURIComponent(q)}&hl=ko&gl=KR&ceid=KR:ko`);
    res.json({ items: feed.items.slice(0, 10) });
});

app.listen(3000, () => console.log('Portal running at http://localhost:3000'));
